- [ ] **Step 1: Create PickerField.jsx**

Create `zuuchmap_app/src/components/PickerField.jsx`:

```javascript
import React, { useState, useEffect, useMemo } from 'react';
import { TouchableOpacity, Text, Modal, View, StyleSheet } from 'react-native';
import { Picker } from '@react-native-picker/picker';
import { Ionicons } from '@expo/vector-icons';
import { spacing, typography, radius, globalStyles, interactions } from '../design/theme';
import { useAppTheme } from '../hooks/useAppTheme';
import { useTranslation } from 'react-i18next';

const PickerField = ({ value, options = [], onSelect, placeholder, error, title }) => {
    const [visible, setVisible] = useState(false);
    const [tempValue, setTempValue] = useState(value);
    const { colors } = useAppTheme();
    const { t } = useTranslation();
    const styles = useMemo(() => createStyles(colors), [colors]);

    useEffect(() => { setTempValue(value); }, [value, visible]);

    const displayValue = options.find(opt =>
        (typeof opt === 'string' ? opt : opt.value) === value
    );
    const displayLabel = displayValue
        ? (typeof displayValue === 'string' ? displayValue : displayValue.label)
        : value || placeholder;

    const handleConfirm = () => {
        onSelect(tempValue);
        setVisible(false);
    };

    return (
        <>
            <TouchableOpacity
                style={[styles.button, error && globalStyles.inputError]}
                onPress={() => setVisible(true)}
                activeOpacity={interactions.activeOpacityLight}
            >
                <Text style={[styles.buttonText, !value && globalStyles.placeholderText]}>
                    {displayLabel}
                </Text>
                <Ionicons name="chevron-down" size={20} color={colors.primary} />
            </TouchableOpacity>

            <Modal visible={visible} transparent animationType="slide" onRequestClose={() => setVisible(false)}>
                <View style={[globalStyles.modalOverlay, { backgroundColor: colors.opacity.overlay }]}>
                    <View style={[globalStyles.modalContent, { backgroundColor: colors.surface, borderTopColor: colors.border.light }]}>
                        <View style={[globalStyles.modalHeader, { borderBottomColor: colors.border.light }]}>
                            <TouchableOpacity onPress={() => setVisible(false)} style={globalStyles.modalButton} activeOpacity={interactions.activeOpacity}>
                                <Text style={[globalStyles.modalButtonText, { color: colors.primary }]}>{t('common.cancel')}</Text>
                            </TouchableOpacity>
                            <Text style={[globalStyles.modalTitle, { color: colors.text.inverse }]}>{title}</Text>
                            <TouchableOpacity onPress={handleConfirm} style={globalStyles.modalButton} activeOpacity={interactions.activeOpacity}>
                                <Text style={[globalStyles.modalButtonText, { color: colors.primary, fontWeight: typography.weight.semibold }]}>{t('common.done')}</Text>
                            </TouchableOpacity>
                        </View>
                        <View style={styles.pickerWrap}>
                            <Picker
                                selectedValue={tempValue}
                                onValueChange={setTempValue}
                                style={[styles.picker, { color: colors.text.inverse }]}
                                itemStyle={[styles.pickerItem, { color: colors.text.inverse }]}
                            >
                                {options.map((opt) => {
                                    const val = typeof opt === 'string' ? opt : opt.value;
                                    const label = typeof opt === 'string' ? opt : opt.label;
                                    return <Picker.Item key={val} label={label} value={val} />;
                                })}
                            </Picker>
                        </View>
                    </View>
                </View>
            </Modal>
        </>
    );
};

const createStyles = (colors) => StyleSheet.create({
    button: {
        flexDirection: 'row',
        justifyContent: 'space-between',
        alignItems: 'center',
        backgroundColor: colors.surface,
        borderWidth: 1,
        borderColor: colors.border.light,
        borderRadius: radius.input,
        paddingHorizontal: spacing.md,
        paddingVertical: spacing.md,
        minHeight: 52,
    },
    buttonText: {
        fontSize: typography.md,
        color: colors.text.inverse,
    },
    pickerWrap: { height: 200 },
    picker: { height: 200, width: '100%' },
    pickerItem: { fontSize: typography.md, height: 200 },
});

export default PickerField;
```
