### Task 1: Post expiry cron

Posts have a `Status.EXPIRED` enum value but nothing sets it. A scheduled cron marks posts `EXPIRED` after 90 days and sends a push notification to the provider.

- [ ] **Step 2: Create post.cron.ts**

Create `zuuchmap_engine/src/post/post.cron.ts`:

```typescript
import { Injectable, Logger } from '@nestjs/common';
import { Cron, CronExpression } from '@nestjs/schedule';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository, LessThan } from 'typeorm';
import { Post } from './entities/post.entity';
import { sendPushNotification } from '../utils/pushNotification';

const EXPIRY_DAYS = 90;

@Injectable()
export class PostCron {
    private readonly logger = new Logger(PostCron.name);

    constructor(
        @InjectRepository(Post)
        private readonly postRepository: Repository<Post>,
    ) {}

    @Cron(CronExpression.EVERY_DAY_AT_MIDNIGHT)
    async expireOldPosts() {
        const cutoff = new Date();
        cutoff.setDate(cutoff.getDate() - EXPIRY_DAYS);

        const posts = await this.postRepository.find({
            where: {
                status: 'ACTIVE' as any,
                approval_status: 'APPROVED' as any,
                date_created: LessThan(cutoff),
            },
            relations: ['user'],
        });

        if (!posts.length) return;

        await this.postRepository.update(
            posts.map(p => p.id),
            { status: 'EXPIRED' as any },
        );

        this.logger.log(`Expired ${posts.length} posts older than ${EXPIRY_DAYS} days`);

        for (const post of posts) {
            if (post.user?.push_token) {
                await sendPushNotification(
                    post.user.push_token,
                    'Зар дууссан',
                    `"${post.title || 'Таны зар'}" ${EXPIRY_DAYS} хоног өнгөрсөн тул дуусгавар болголоо.`,
                    { postId: post.id },
                ).catch(() => {});
            }
        }
    }
}
```

```typescript
import { Entity, PrimaryGeneratedColumn, Column, ManyToOne, JoinColumn, CreateDateColumn, Index } from 'typeorm';
import { User } from '../user/entities/user.entity';
import { Post } from '../post/entities/post.entity';

@Entity('messages')
@Index(['from_user', 'to_user', 'post'])
export class Message {
    @PrimaryGeneratedColumn()
    id: number;

    @ManyToOne(() => User, { onDelete: 'CASCADE' })
    @JoinColumn({ name: 'from_user_id' })
    from_user: User;

    @ManyToOne(() => User, { onDelete: 'CASCADE' })
    @JoinColumn({ name: 'to_user_id' })
    to_user: User;

    @ManyToOne(() => Post, { onDelete: 'CASCADE' })
    @JoinColumn({ name: 'post_id' })
    post: Post;

    @Column({ type: 'text' })
    text: string;

    @Column({ default: false })
    read: boolean;

    @CreateDateColumn()
    date_created: Date;
}
```

```typescript
import { Injectable, NotFoundException } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository } from 'typeorm';
import { Message } from './message.entity';
import { CreateMessageDto } from './dto/create-message.dto';

@Injectable()
export class MessageService {
    constructor(
        @InjectRepository(Message)
        private readonly messageRepository: Repository<Message>,
    ) {}

    async send(fromUserId: string, dto: CreateMessageDto): Promise<Message> {
        const msg = this.messageRepository.create({
            from_user: { id: fromUserId } as any,
            to_user: { id: dto.to_user_id } as any,
            post: { id: dto.post_id } as any,
            text: dto.text,
        });
        return this.messageRepository.save(msg);
    }

    async getConversations(userId: string): Promise<object[]> {
        const raw = await this.messageRepository
            .createQueryBuilder('m')
            .leftJoinAndSelect('m.from_user', 'fu')
            .leftJoinAndSelect('m.to_user', 'tu')
            .leftJoinAndSelect('m.post', 'p')
            .where('m.from_user_id = :uid OR m.to_user_id = :uid', { uid: userId })
            .orderBy('m.date_created', 'DESC')
            .getMany();

        // Group by (post_id, other_user_id), keep latest message per conversation
        const seen = new Set<string>();
        const conversations = [];
        for (const msg of raw) {
            const otherId = msg.from_user.id === userId ? msg.to_user.id : msg.from_user.id;
            const key = `${msg.post.id}:${otherId}`;
            if (!seen.has(key)) {
                seen.add(key);
                const unread = await this.messageRepository.count({
                    where: { to_user: { id: userId }, post: { id: msg.post.id }, from_user: { id: otherId }, read: false },
                });
                conversations.push({
                    post_id: msg.post.id,
                    post_title: msg.post.title,
                    other_user: { id: otherId, given_name: (msg.from_user.id === userId ? msg.to_user : msg.from_user).given_name },
                    last_message: msg.text,
                    last_date: msg.date_created,
                    unread_count: unread,
                });
            }
        }
        return conversations;
    }

    async getMessages(userId: string, postId: number, otherUserId: string): Promise<Message[]> {
        const msgs = await this.messageRepository.find({
            where: [
                { from_user: { id: userId }, to_user: { id: otherUserId }, post: { id: postId } },
                { from_user: { id: otherUserId }, to_user: { id: userId }, post: { id: postId } },
            ],
            relations: ['from_user', 'to_user'],
            order: { date_created: 'ASC' },
        });
        // Mark received messages as read
        const unread = msgs.filter(m => m.to_user.id === userId && !m.read);
        if (unread.length) {
            await this.messageRepository.update(unread.map(m => m.id), { read: true });
        }
        return msgs;
    }
}
```

Create `zuuchmap_engine/src/message/message.controller.ts`:

```typescript
import { Controller, Post, Get, Body, Param, ParseIntPipe, Req, UseGuards, HttpException, HttpStatus } from '@nestjs/common';
import { MessageService } from './message.service';
import { CreateMessageDto } from './dto/create-message.dto';
import { JwtAuthGuard } from '../auth/jwt-auth.guard';
import { EventsGateway } from '../events/events.gateway';

@Controller('messages')
@UseGuards(JwtAuthGuard)
export class MessageController {
    constructor(
        private readonly messageService: MessageService,
        private readonly events: EventsGateway,
    ) {}

    @Post()
    async send(@Req() req, @Body() dto: CreateMessageDto) {
        try {
            const msg = await this.messageService.send(req.user.id, dto);
            this.events.emitMessageReceived(dto.to_user_id, {
                from_user_id: req.user.id,
                post_id: dto.post_id,
                text: dto.text,
                date_created: msg.date_created,
            });
            return { success: true, data: msg };
        } catch (error) {
            throw new HttpException(error.message || 'Failed to send message', error.status || HttpStatus.INTERNAL_SERVER_ERROR);
        }
    }

    @Get('conversations')
    async getConversations(@Req() req) {
        try {
            const data = await this.messageService.getConversations(req.user.id);
            return { success: true, data };
        } catch (error) {
            throw new HttpException(error.message || 'Failed to get conversations', HttpStatus.INTERNAL_SERVER_ERROR);
        }
    }

    @Get('conversation/:postId/:otherUserId')
    async getMessages(
        @Req() req,
        @Param('postId', ParseIntPipe) postId: number,
        @Param('otherUserId') otherUserId: string,
    ) {
        try {
            const data = await this.messageService.getMessages(req.user.id, postId, otherUserId);
            return { success: true, data };
        } catch (error) {
            throw new HttpException(error.message || 'Failed to get messages', HttpStatus.INTERNAL_SERVER_ERROR);
        }
    }
}
```

```javascript
import React, { useState, useEffect, useCallback, useRef } from 'react';
import { View, Text, FlatList, TextInput, TouchableOpacity, StyleSheet, KeyboardAvoidingView, Platform } from 'react-native';
import { Ionicons } from '@expo/vector-icons';
import { useTranslation } from 'react-i18next';
import { ScreenLayout } from '../../components';
import { spacing, typography, radius } from '../../design/theme';
import { useAppTheme } from '../../hooks/useAppTheme';
import messageService from '../../services/api/messageService';
import { getUserId } from '../../services/api/authHelpers';

const ConversationScreen = ({ route, navigation }) => {
    const { postId, otherUserId, title } = route.params;
    const { t } = useTranslation();
    const { colors } = useAppTheme();
    const [messages, setMessages] = useState([]);
    const [text, setText] = useState('');
    const [sending, setSending] = useState(false);
    const [myId, setMyId] = useState(null);
    const listRef = useRef(null);

    useEffect(() => { getUserId().then(setMyId); }, []);

    const load = useCallback(async () => {
        try {
            const res = await messageService.getMessages(postId, otherUserId);
            setMessages(res.data?.data ?? []);
        } catch {}
    }, [postId, otherUserId]);

    useEffect(() => { load(); }, [load]);

    const send = async () => {
        const trimmed = text.trim();
        if (!trimmed || sending) return;
        setSending(true);
        try {
            await messageService.send(otherUserId, postId, trimmed);
            setText('');
            await load();
            listRef.current?.scrollToEnd({ animated: true });
        } catch {}
        setSending(false);
    };

    const renderItem = ({ item }) => {
        const isMine = item.from_user?.id === myId;
        return (
            <View style={[styles.bubble, isMine ? styles.bubbleMine : styles.bubbleTheirs]}>
                <View style={[
                    styles.bubbleInner,
                    { backgroundColor: isMine ? colors.primary : colors.surface }
                ]}>
                    <Text style={[styles.bubbleText, { color: isMine ? '#fff' : colors.text.inverse }]}>
                        {item.text}
                    </Text>
                </View>
            </View>
        );
    };

    return (
        <ScreenLayout title={title} showBack onBack={() => navigation.goBack()}>
            <KeyboardAvoidingView
                style={{ flex: 1 }}
                behavior={Platform.OS === 'ios' ? 'padding' : undefined}
                keyboardVerticalOffset={88}
            >
                <FlatList
                    ref={listRef}
                    data={messages}
                    keyExtractor={item => String(item.id)}
                    renderItem={renderItem}
                    contentContainerStyle={styles.list}
                    onContentSizeChange={() => listRef.current?.scrollToEnd()}
                />
                <View style={[styles.inputRow, { borderTopColor: colors.border.light, backgroundColor: colors.surface }]}>
                    <TextInput
                        style={[styles.input, { backgroundColor: colors.background, color: colors.text.inverse, borderColor: colors.border.light }]}
                        value={text}
                        onChangeText={setText}
                        placeholder={t('messages.inputPlaceholder')}
                        placeholderTextColor={colors.text.secondary}
                        multiline
                        maxLength={1000}
                    />
                    <TouchableOpacity
                        style={[styles.sendBtn, { backgroundColor: colors.primary }]}
                        onPress={send}
                        disabled={sending || !text.trim()}
                        activeOpacity={0.7}
                    >
                        <Ionicons name="send" size={18} color="#fff" />
                    </TouchableOpacity>
                </View>
            </KeyboardAvoidingView>
        </ScreenLayout>
    );
};

const styles = StyleSheet.create({
    list: { padding: spacing.md, paddingBottom: spacing.xl },
    bubble: { marginBottom: spacing.xs },
    bubbleMine: { alignItems: 'flex-end' },
    bubbleTheirs: { alignItems: 'flex-start' },
    bubbleInner: { maxWidth: '75%', borderRadius: radius.xl, padding: spacing.sm },
    bubbleText: { fontSize: typography.sm, lineHeight: typography.sm * 1.5 },
    inputRow: { flexDirection: 'row', alignItems: 'flex-end', padding: spacing.sm, borderTopWidth: StyleSheet.hairlineWidth, gap: spacing.sm },

    sendBtn: { width: 40, height: 40, borderRadius: 20, justifyContent: 'center', alignItems: 'center' },
});

export default ConversationScreen;
```
